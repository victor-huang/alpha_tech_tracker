```
  --log-level DEBUG
  --window M1 09:30 3        # morning opening range (15-min)
  --window A1 10:00 3        # late-morning window
  --window A2 13:15 1        # early afternoon window
  --window A3 15:15 1        # close window
  --morning-split 100        # full capital to M1 first
  --reversal                 # reversal re-entries enabled
  --bearish-reentry          # bearish re-entry enabled
  --bullish-reentry          # bullish re-entry enabled
  --doubledown               # double-down add-on enabled
  --doubledown-start 10      # DD fires 10 min after OR close
  --rank-weighted-sizing 60 40   # rank-1: 60%, rank-2: 40%
  --top 2                    # top 2 tickers per day
  --capital 10000            # $10k starting capital
  --mock-trade-execution     # simulated fills
  --feed sip                 # Alpaca SIP feed
  --trade-type stock|options # varies per run
  --replay-date YYYY-MM-DD
```
